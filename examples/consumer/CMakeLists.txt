cmake_minimum_required(VERSION 3.24)
project(MeshvaleInstalledConsumer LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(MeshvaleGeometry CONFIG REQUIRED)
file(READ "${MeshvaleGeometry_REPORT_SCHEMA}" meshvale_report_schema)
string(JSON meshvale_report_dialect GET "${meshvale_report_schema}" "$schema")
if(NOT meshvale_report_dialect STREQUAL "https://json-schema.org/draft/2020-12/schema")
  message(FATAL_ERROR "Installed report schema has an unexpected dialect")
endif()
message(STATUS "Installed declarative report schema found without Python")
add_executable(meshvale_consumer main.cpp)
target_link_libraries(meshvale_consumer PRIVATE meshvale::geometry)
foreach(consumer_header IN ITEMS attributes mesh topology)
  foreach(consumer_extension IN ITEMS h)
    set(consumer_probe "${CMAKE_CURRENT_BINARY_DIR}/${consumer_header}_${consumer_extension}.cpp")
    file(WRITE "${consumer_probe}"
      "#include <meshvale/geometry/${consumer_header}.${consumer_extension}>\n#include <meshvale/geometry/${consumer_header}.${consumer_extension}>\n")
    target_sources(meshvale_consumer PRIVATE "${consumer_probe}")
  endforeach()
endforeach()
if(MSVC)
  target_compile_options(meshvale_consumer PRIVATE /W4 /WX)
else()
  target_compile_options(meshvale_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME installed_consumer COMMAND meshvale_consumer)

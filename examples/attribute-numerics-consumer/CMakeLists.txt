cmake_minimum_required(VERSION 3.24)
project(MeshvaleAttributeNumericsConsumer LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(MeshvaleGeometry CONFIG REQUIRED)
add_executable(attribute_numerics_consumer main.cpp)
target_link_libraries(attribute_numerics_consumer PRIVATE meshvale::attribute_numerics)
if(MSVC)
  target_compile_options(attribute_numerics_consumer PRIVATE /W4 /WX /fp:strict)
else()
  target_compile_options(attribute_numerics_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math)
endif()
enable_testing()
add_test(NAME installed_attribute_numerics COMMAND attribute_numerics_consumer)
